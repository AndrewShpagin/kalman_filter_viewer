target_compile_features(flight_kalman INTERFACE cxx_std_17)

add_library(flight_velocity INTERFACE)
target_include_directories(flight_velocity INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(flight_velocity INTERFACE cxx_std_17)

add_executable(velocity_filter_csv velocity_filter_csv.cpp)
target_link_libraries(velocity_filter_csv PRIVATE flight_velocity)
set_target_properties(velocity_filter_csv PROPERTIES RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)
if(MSVC)
    target_compile_options(velocity_filter_csv PRIVATE /W4 /utf-8)
elseif(MINGW)
    target_link_options(velocity_filter_csv PRIVATE -municode)
else()
    target_compile_options(velocity_filter_csv PRIVATE -Wall -Wextra -Wpedantic)
endif()
